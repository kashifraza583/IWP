<script>
$(document).ready(function(){

    $("#check").click(function(){
        var d = new Date($("#date").val());
        var day = d.getDay();

        if(day == 0 || day == 6)
            alert("Weekend");
        else
            alert("Weekday");
    });

});
</script>
